def run(playwright: Playwright) -> None:
    browser = playwright.chromium.launch(headless=False)
    context = browser.new_context()
    page.get_by_text("签约管理").click()
    page.get_by_text("签约", exact=True).first.click()
    page.get_by_text("未启动", exact=True).click()
    page.get_by_label("签约项目").click()
    page.locator(".ml-15 > div").first.click()
    page.get_by_text("启动", exact=True).click()
    page.get_by_text("取消").click()

    # ---------------------
    context.close()
    browser.close()
